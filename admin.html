document .querySelectorAll("[data-share]") .forEach((button) => { button.addEventListener("click", async () => { const quizId = button.dataset.share; // quiz.html এবং admin.html একই folder-এ থাকলে const quizUrl = new URL("quiz.html", window.location.href); quizUrl.searchParams.set("id", quizId); const finalUrl = quizUrl.href; try { await navigator.clipboard.writeText(finalUrl); showMessage( "Quiz link copied! এখন বন্ধুদের পাঠাতে পারো।" ); } catch (error) { console.error("Clipboard error:", error); // Clipboard কাজ না করলে সরাসরি link দেখাবে window.prompt( "এই Quiz link-টি copy করো:", finalUrl ); } }); });
